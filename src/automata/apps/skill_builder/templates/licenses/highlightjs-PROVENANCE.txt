Reproduce in an empty disposable build directory, using npm and esbuild 0.25.12
(already cached at the original build; no package installation needed):

  npm pack https://registry.npmjs.org/highlight.js/-/highlight.js-11.11.1.tgz \
    --offline --ignore-scripts
  tar -xzf highlight.js-11.11.1.tgz

Create highlight-entry.js with these exact contents:

import hljs from './package/lib/core.js';
import javascript from './package/lib/languages/javascript.js';
import json from './package/lib/languages/json.js';
import bash from './package/lib/languages/bash.js';
import python from './package/lib/languages/python.js';
import yaml from './package/lib/languages/yaml.js';
for (const [name, grammar] of Object.entries({javascript, json, bash, python, yaml})) {
  hljs.registerLanguage(name, grammar);
}
export default hljs;

Then run (esbuild is the pinned executable):

  esbuild highlight-entry.js --bundle --minify --format=esm --platform=browser \
    --target=es2022 \
    '--banner:js=/*! Highlight.js 11.11.1 | BSD-3-Clause | See ../licenses/highlightjs-LICENSE.txt */' \
    --outfile=highlight.js

Compare SHA-256 values above, copy highlight.js into templates/lib/ and copy
package/LICENSE unchanged into templates/licenses/highlightjs-LICENSE.txt.
Fetching absent build inputs requires separate download authorization; serving
this already-bundled site needs none. Do not replace the other reviewed bundles.
